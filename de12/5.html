プログラミング

今回はpythonを使って「シーン別レストラン提案」を作ってみました。
はじめにchatGPTのAPIを利用し,行きたいレストランのジャンル(和食、洋食、中華など)とおすすめのレストランの名前が出てくるようなソースコードを書いてもらいました。
そしてそのソースコードに、予算・レストランのあるエリア・どんな人と行くか・みなとみらいキャンパスからの移動時間の項目を追加し、レストランも実際にあるお店に書き換えました。

ソースコード

1import random
2#レストランの候補
3restaurants = [
4    {"name": "ブリル飯店", "genre": "中華", "price": 4000, "place":"横浜", "people":"友達or家族","traveltime":16},
5    {"name": "ごはんすすむ", "genre": "和食", "price": 2000, "place":"横浜", "people":"一人","traveltime":16},
6    {"name": "つばめグリル", "genre": "洋食", "price": 3000, "place":"横浜","people":"友達or恋人","traveltime":18},
7    {"name": "TSUBAKI食堂", "genre": "中華", "price": 2000, "place":"みなとみらい","people":"友達or恋人","traveltime":18},
8    {"name": "橙家", "genre": "和食", "price": 3000, "place":"みなとみらい","people":"友達or家族","traveltime":14},
9    {"name": "茶茶　このか", "genre": "和食", "price": 6000, "place":"渋谷","people":"恋人or家族","traveltime":53},
10    {"name": "君のハンバーグを食べたい", "genre": "洋食", "price": 2000, "place":"渋谷","people":"友達or一人","traveltime":53},
11    {"name": "Italianbar Attachment", "genre": "洋食", "price": 2000, "place":"新宿","people":"友達or恋人","traveltime":60},
12    {"name": "なきざかな", "genre": "和食", "price": 5000, "place":"新宿","people":"友達or家族","traveltime":56},
13]
14#ジャンル、予算、場所、誰と行くかの選択肢
15genre = input("食べたいジャンル（和食・洋食・中華・指定なし）：")
16budget = int(input("予算の上限（円）："))
17place = input("場所（横浜・みなとみらい・渋谷・新宿）:")
18people = input("誰と行くか(友達・恋人・家族・一人):")
19
20
21candidates = [
22    restaurant
23    for restaurant in restaurants
24    if restaurant["price"] <= budget
25    and (genre == "指定なし" or restaurant["genre"] == genre)
26    and restaurant["place"] == place
27    and people in restaurant["people"]
28]
29
30#choiceでお店のデータと移動時間を連動させる
31if candidates:
32    choice = random.choice(candidates)
33
34    print(f"おすすめは「{choice['name']}」です！")
35    print(f"ジャンル：{choice['genre']}")
36    print(f"予算：約{choice['price']}円")
37    print(f"場所：{choice['place']}")
38    print(f"移動時間：約{choice['traveltime']}分です！")
39
40else:
41    print("条件に合うお店が見つかりませんでした。")
